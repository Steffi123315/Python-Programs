#REVERSE UPPER CASE TO LOWER CASE AND VICE VERSA
s=input('enter string:')
out=''
for i in s:
  if "A"<=i<="Z":
    out+=chr(ord(i)+32)
  elif'a'<=i<='z':
    out+=chr(ord(i)-32)
  else:
    out+=i
print(out)
